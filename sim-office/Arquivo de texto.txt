function pointToGrid(x, z) { // converte coords reais para grid
    return [
        Math.floor((x + 0.5*GRID_SIZE)/GRID_SIZE),
        Math.floor((z + 0.5*GRID_SIZE)/GRID_SIZE)
    ];
}
function gridToPoint(gx, gz) {
    return [
        gx * GRID_SIZE,
        gz * GRID_SIZE
    ];
}

function addWall(x, y, z, w, h, d, doorList=[]) {
    // portas são furos no mesh
    const shape = new THREE.Shape();
    shape.moveTo(0,0);
    shape.lineTo(w,0);
    shape.lineTo(w,h);
    shape.lineTo(0,h);
    shape.lineTo(0,0);
    doorList.forEach(door => {
        // cada porta é um "furo" (3x2m)
        let dh=2.2, dw=1.45;
        let dx = door.x, dz=door.z||0, dy=0;
        let doorShape = new THREE.Path();
        doorShape.moveTo(dx,dy);
        doorShape.lineTo(dx+dw,dy);
        doorShape.lineTo(dx+dw,dy+dh);
        doorShape.lineTo(dx,dy+dh);
        doorShape.lineTo(dx,dy);
        shape.holes.push(doorShape);
    });
    let geo = new THREE.ExtrudeGeometry(shape, { depth: d, bevelEnabled: false });
    let mat = new THREE.MeshLambertMaterial({ color: 0xeee8da, side: THREE.DoubleSide });
    let wall = new THREE.Mesh(geo, mat);
    wall.position.set(x, y, z);
    wall.castShadow = true;
    wall.receiveShadow = true;
    return wall;
}

// Mapeando zonas walkable (grelha simples)
function gerarNavGrid() {
    let nx = Math.floor((office.w)/GRID_SIZE); // Adjusted bounds slightly
    let nz = Math.floor((office.d)/GRID_SIZE); // Adjusted bounds slightly
    navGrid = [];
    for(let x=0; x<=nx; x++){
        navGrid[x] = [];
        for(let z=0; z<=nz; z++){
            let qX = x*GRID_SIZE + 0.5*GRID_SIZE; // Use center of grid cell
            let qZ = z*GRID_SIZE + 0.5*GRID_SIZE; // Use center of grid cell

            // Teste simples: está DENTRO de uma sala (não na parede)?
            let walk = false;
            let wallThicknessMargin = office.wallT * 0.6; // Small margin to avoid being inside walls

            for (let key in office.rooms) {
                const r = office.rooms[key];
                if (qX > r.x + wallThicknessMargin &&
                    qX < r.x + r.w - wallThicknessMargin &&
                    qZ > r.z + wallThicknessMargin &&
                    qZ < r.z + r.d - wallThicknessMargin ){
                    walk = true;
                    break; // Found a room it's inside
                }
            }
            navGrid[x][z] = walk;
        }
    }
    // abrir os "furos" das portas (making area around door walkable)
    function portaToGridCoords(roomX, roomZ, doorX, doorZ){
        // Calculate world coordinates of door center more accurately
        // Assuming doors are centered on the wall thickness
        let worldX, worldZ;
        if (doorZ === 0) { // Door on front wall (relative to room)
             worldX = roomX + doorX + 1.45/2; // center of door width
             worldZ = roomZ + office.wallT/2;
        } else if (doorX === 0) { // Door on left wall (relative to room)
             worldX = roomX + office.wallT/2;
             worldZ = roomZ + doorZ + 1.45/2; // center of door height (use width for calculation)
        } else { // Add cases for other walls if needed
            worldX = roomX + doorX;
            worldZ = roomZ + doorZ;
        }
        return pointToGrid(worldX, worldZ);
    }

    for (let key in office.rooms) {
        const room = office.rooms[key];
        room.doors.forEach(porta=>{
            let [doorGX, doorGZ] = portaToGridCoords(room.x, room.z, porta.x, porta.z || 0);
             // Make a small area around the door grid coords walkable
            for(let i=-1; i<=1; i++){
                for(let j=-1; j<=1; j++){
                    let gx = doorGX + i;
                    let gz = doorGZ + j;
                    if (gx >= 0 && gx < navGrid.length && gz >= 0 && navGrid[gx] && gz < navGrid[gx].length) {
                         // Only set to true if it wasn't already false (don't overwrite existing walls)
                         // Actually, let's just force it true for doors
                         navGrid[gx][gz] = true;
                    }
                }
            }
        });
    }
}

// --- Pathfinding muito básico (A*) em grid walkable (simplificado) ---
function AStarPath(startGrid, endGrid) {
    // Heuristic function (Manhattan distance)
    function h(a, b) { return Math.abs(a[0] - b[0]) + Math.abs(a[1] - b[1]); }

    const openSet = new Map(); // Map: "x,y" -> { node: [x,y], gScore: number, fScore: number }
    const cameFrom = new Map(); // Map: "x,y" -> [prevX, prevY]

    const startKey = startGrid.join(",");
    openSet.set(startKey, { node: startGrid, gScore: 0, fScore: h(startGrid, endGrid) });

    const closedSet = new Set(); // Set: "x,y"

    while (openSet.size > 0) {
        // Find node in openSet with the lowest fScore
        let currentKey = null;
        let lowestFScore = Infinity;
        for (const [key, data] of openSet.entries()) {
            if (data.fScore < lowestFScore) {
                lowestFScore = data.fScore;
                currentKey = key;
            }
        }

        const currentData = openSet.get(currentKey);
        const currentNode = currentData.node;

        // Goal reached?
        if (currentNode[0] === endGrid[0] && currentNode[1] === endGrid[1]) {
            // Reconstruct path
            const path = [currentNode];
            let tempKey = currentKey;
            while (cameFrom.has(tempKey)) {
                const prevNode = cameFrom.get(tempKey);
                path.push(prevNode);
                tempKey = prevNode.join(",");
            }
            return path.reverse();
        }

        openSet.delete(currentKey);
        closedSet.add(currentKey);

        // Explore neighbors (4 directions)
        for (const [dx, dz] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
            const neighborNode = [currentNode[0] + dx, currentNode[1] + dz];
            const neighborKey = neighborNode.join(",");

            // Check bounds and walkability
            if (neighborNode[0] < 0 || neighborNode[0] >= navGrid.length ||
                neighborNode[1] < 0 || !navGrid[neighborNode[0]] || neighborNode[1] >= navGrid[neighborNode[0]].length ||
                !navGrid[neighborNode[0]][neighborNode[1]] || // Check if walkable
                closedSet.has(neighborKey)) {
                continue; // Skip invalid or already evaluated neighbors
            }

            const tentativeGScore = currentData.gScore + 1; // Cost is 1 per step

            let neighborData = openSet.get(neighborKey);
            if (!neighborData || tentativeGScore < neighborData.gScore) {
                 // This path to neighbor is better than any previous one. Record it!
                cameFrom.set(neighborKey, currentNode);
                 if (!neighborData) {
                     neighborData = { node: neighborNode, gScore: 0, fScore: 0 }; // Create if new
                 }
                 neighborData.gScore = tentativeGScore;
                 neighborData.fScore = tentativeGScore + h(neighborNode, endGrid);
                 openSet.set(neighborKey, neighborData); // Add/Update in openSet
            }
        }
    }

    // Open set is empty but goal was never reached
    console.warn("A* Path not found from", startGrid, "to", endGrid);
    return null;
}


function findNavTo(startPos, endPos){
    let st = pointToGrid(startPos[0], startPos[2]);
    let end = pointToGrid(endPos[0], endPos[2]);

    // Ensure start/end points are actually walkable, if not, find nearest walkable
     if (!navGrid[st[0]] || !navGrid[st[0]][st[1]]) {
         console.warn("Start point not walkable, finding nearest for", st);
         st = findNearestWalkable(st);
         if (!st) return null; // No walkable near start
     }
     if (!navGrid[end[0]] || !navGrid[end[0]][end[1]]) {
        console.warn("End point not walkable, finding nearest for", end);
        end = findNearestWalkable(end);
        if (!end) return null; // No walkable near end
     }

    let path = AStarPath(st, end);
    if(!path) return null;

    // Convert grid path back to world points (centers of grid cells)
    return path.map(([gx,gz])=>{
        let p = gridToPoint(gx,gz);
        return [p[0] + GRID_SIZE/2, startPos[1], p[1] + GRID_SIZE/2]; // Use Y from original startPos
    });
}

// Helper to find the nearest walkable grid cell (simple spiral search)
function findNearestWalkable(startGrid) {
    const maxSearchRadius = 5; // Limit search to avoid infinite loop
    if (navGrid[startGrid[0]] && navGrid[startGrid[0]][startGrid[1]]) {
        return startGrid; // Already walkable
    }
    for (let radius = 1; radius <= maxSearchRadius; radius++) {
        for (let dx = -radius; dx <= radius; dx++) {
            for (let dz = -radius; dz <= radius; dz++) {
                if (Math.abs(dx) !== radius && Math.abs(dz) !== radius) continue; // Only check perimeter
                let nx = startGrid[0] + dx;
                let nz = startGrid[1] + dz;
                if (nx >= 0 && nx < navGrid.length && nz >= 0 && navGrid[nx] && nz < navGrid[nx].length && navGrid[nx][nz]) {
                    return [nx, nz];
                }
            }
        }
    }
    console.error("Could not find walkable near", startGrid);
    return null; // None found within radius
}

        l.position.set(0.09*a,0.18,0.03); worker.add(l);
    }
    // informação lógica
    worker.userData = {
        id: index,
        nome: "Dev_"+(index+1),
        pos: [...pos], // pos grid real (x,y,z) - Initial position
        direcao: dir, // Initial facing direction if relevant
        rot: dir>0?Math.PI/12:-Math.PI/12, // Initial rotation? Seems unused
        hunger: 80+Math.random()*18,
        sleep: 88+Math.random()*10,
        toilet: 94+Math.random()*4,
        marker: null,
        path: [],
        state: "trabalhando", // Start at workstation
        seat: [...pos], // Store the assigned work seat position
        obj: worker,
        destino: null, // Current target position object {pos: [x,y,z], ...}
        actionTimeout: 0,
        markerSel: null // mesh marcador seleção
    };
    worker.position.set(...pos);
    worker.rotation.y = (dir === 1) ? Math.PI : 0; // Face desk initially
    worker.castShadow = true;
    // Make children cast shadow too
    worker.traverse(child => {
        if (child.isMesh) {
            child.castShadow = true;
            child.receiveShadow = true; // Meshes can also receive shadows
        }
    });
    workers_pool.push(worker);
    scene.add(worker);
    return worker;
}

function addMarcador(worker, color="#ec3d41") {
    // Using a simple 3D sprite instead of DOM element for better integration
    const canvas = document.createElement('canvas');
    canvas.width = 64;
    canvas.height = 64;
    const context = canvas.getContext('2d');

    context.beginPath();
    context.arc(32, 32, 28, 0, Math.PI * 2);
    context.fill();
    context.strokeStyle = '#900'; // Border color from original style
    context.lineWidth = 4; // Border width from original style
    context.stroke();

    const texture = new THREE.CanvasTexture(canvas);
    const spriteMaterial = new THREE.SpriteMaterial({ map: texture });
    const sprite = new THREE.Sprite(spriteMaterial);
    sprite.scale.set(0.3, 0.3, 1); // Adjust size as needed
    sprite.position.set(0, 1.6, 0); // Position above head

    worker.add(sprite); // Add sprite to the worker group
    worker.userData.marker = sprite; // Store reference to the sprite
    sprite.visible = false; // Initially hidden, show based on state? Or always show? Let's start visible
}

// ---- AGENTES/IA ----
const WORKER_SPEED = 1.8; // Units per second
const ACTION_TIME_HUNGER = 4.0; // Base time in seconds
const ACTION_TIME_TOILET = 3.5;
const ACTION_TIME_SLEEP = 6.0;

function updateWorkerLogical(w, dt) {
    let ud = w.userData;

     // Update needs only when working or moving (not during actions)
     if (ud.state === 'trabalhando' || ud.state === 'voltando' || ud.state.startsWith('indo_')) {
        // Reduz Ao acaso (mas lento) as barras de necessidades
        ud.hunger -= (0.6 + 1.0 * Math.random()) * dt; // Faster decrease
        ud.sleep  -= (0.5 + 0.8 * Math.random()) * dt;
        ud.toilet -= (0.4 + 0.9 * Math.random()) * dt; // Slightly faster decrease
     }

     // Clamp needs to minimum 0
     ud.hunger = Math.max(0, ud.hunger);
     ud.sleep = Math.max(0, ud.sleep);
     ud.toilet = Math.max(0, ud.toilet);

    // --- State Machine Logic ---
    if (ud.state === "trabalhando") {
         // Check needs
        if (ud.hunger <= 10 && !ud.state.startsWith('indo_')) // Threshold to seek food
            trocar_state(w, 'indo_cozinha');
        else if (ud.toilet <= 15 && !ud.state.startsWith('indo_')) // Threshold
            trocar_state(w, 'indo_banheiro');
        else if (ud.sleep <= 20 && !ud.state.startsWith('indo_')) // Threshold
            trocar_state(w,'indo_descansar');
        else {
            // Perform idle working animation (slight movement)
            w.position.x = ud.seat[0] + 0.009*Math.sin(performance.now()/321 + ud.id);
            w.position.z = ud.seat[2] + 0.008*Math.cos(performance.now()/317 + ud.id);
             w.rotation.y = (ud.direcao === 1) ? Math.PI : 0; // Ensure facing desk
        }

    } else if (ud.state === "indo_cozinha") {
        if (!ud.destino) {
            // Find available kitchen seat
             let dest = cozinhaAssentos.find(seat => !seat.ocupado);
            if (dest) {
                dest.ocupado = true; // Mark as occupied immediately
                ud.destino = dest;
                ud.path = findNavTo(
                    [w.position.x, w.position.y, w.position.z],
                    [dest.pos[0], w.position.y, dest.pos[2]] // Use current Y level
                );
                 if (!ud.path) { // Pathfinding failed, release seat and go back
                     console.error(`Path to kitchen failed for ${ud.nome}`);
                     dest.ocupado = false;
                     ud.destino = null;
                     trocar_state(w, 'voltando');
                 }
            } else {
                // No seat available, maybe wait or go back? For now, go back.
                trocar_state(w, 'voltando');
            }
        } else {
            moveWorkerAlongPath(w, dt, "comendo");
        }

    } else if (ud.state === "comendo") {
        // Sit and 'eat' - replenish hunger over time? Or instantly?
        // Let's do instant + timeout
        if (ud.actionTimeout <= 0) { // Just started eating
            ud.hunger = 100;
        }
        ud.actionTimeout -= dt;
        if (ud.actionTimeout <= 0) {
             if (ud.destino) {
                 ud.destino.ocupado = false; // Release the seat
             }
             ud.destino = null;
             ud.path = [];
             trocar_state(w, 'voltando');
        }

    } else if (ud.state === "indo_banheiro") {
         if (!ud.destino) {
             let cab = banheirosCabines.find(c => !c.ocupado);
             if (cab) {
                 cab.ocupado = true;
                 ud.destino = cab;
                 ud.path = findNavTo(
                     [w.position.x, w.position.y, w.position.z],
                     [cab.pos[0], w.position.y, cab.pos[2]]
                 );
                 if (!ud.path) {
                    console.error(`Path to bathroom failed for ${ud.nome}`);
                    cab.ocupado = false;
                    ud.destino = null;
                    trocar_state(w, 'voltando');
                }
             } else {
                 trocar_state(w, 'voltando'); // No cabin free
             }
         } else {
             moveWorkerAlongPath(w, dt, "usando_banheiro");
         }

    } else if (ud.state === "usando_banheiro") {
        if (ud.actionTimeout <= 0) { // Just started
            ud.toilet = 100;
        }
        ud.actionTimeout -= dt;
        if (ud.actionTimeout <= 0) {
             if (ud.destino) {
                 ud.destino.ocupado = false; // Release cabin
             }
             ud.destino = null;
             ud.path = [];
             trocar_state(w, 'voltando');
        }

    } else if (ud.state === "indo_descansar") {
        if(!ud.destino){
            let dest = biblioSeats.find(b => !b.ocupado);
            if(dest){
                 dest.ocupado = true;
                 ud.destino = dest;
                 ud.path = findNavTo(
                     [w.position.x, w.position.y, w.position.z],
                     [dest.pos[0], w.position.y, dest.pos[2]]
                 );
                  if (!ud.path) {
                    console.error(`Path to rest area failed for ${ud.nome}`);
                    dest.ocupado = false;
                    ud.destino = null;
                    trocar_state(w, 'voltando');
                  }
            } else {
                 trocar_state(w, 'voltando'); // No sofa free
            }
        } else {
            moveWorkerAlongPath(w, dt, "descansando");
        }

    } else if (ud.state === "descansando") {
        if (ud.actionTimeout <= 0) { // Just started
             ud.sleep = 100;
        }
        ud.actionTimeout -= dt;
        if(ud.actionTimeout <= 0){
             if (ud.destino) {
                 ud.destino.ocupado = false; // Release sofa
             }
             ud.destino = null;
             ud.path = [];
             trocar_state(w,'voltando');
        }

    } else if (ud.state === "voltando") {
        // Go back to the assigned workstation seat
        if (!ud.path || ud.path.length === 0) {
            ud.path = findNavTo([w.position.x, w.position.y, w.position.z], ud.seat);
             if (!ud.path) {
                 console.error(`Path back to seat failed for ${ud.nome}. Trying again next frame.`);
                 // Stay in 'voltando' state, will retry pathfinding next frame
                 return;
             }
        }
        // Check if already very close to seat to avoid pathfinding loop if target is slightly off
         const seatVec = new THREE.Vector3(...ud.seat);
         if (w.position.distanceTo(seatVec) < 0.2) {
             w.position.set(...ud.seat); // Snap to exact seat position
             trocar_state(w, "trabalhando");
         } else {
             moveWorkerAlongPath(w, dt, "trabalhando");
         }
    }
}


function trocar_state(w, novo){
    let ud = w.userData;
    // console.log(`${ud.nome} changing state from ${ud.state} to ${novo}`); // Debug state changes
    ud.state = novo;
    ud.path = []; // Clear path on state change
    ud.destino = null; // Clear destination unless explicitly set by new state logic

    // Set action timeouts for states that involve waiting
    switch(novo){
        case "comendo":         ud.actionTimeout = ACTION_TIME_HUNGER + Math.random()*1.5; break;
        case "usando_banheiro": ud.actionTimeout = ACTION_TIME_TOILET + Math.random()*1.0; break;
        case "descansando":     ud.actionTimeout = ACTION_TIME_SLEEP + Math.random()*2.0; break;
        default:                ud.actionTimeout = 0; break; // No timeout for moving/working
    }

    // Update HUD if this worker is selected
    if (selectedWorker === w) updateHUD();
}

// Movimento automático: segue o path atual
function moveWorkerAlongPath(w, dt, fimState) {
    let ud = w.userData;
    if (!ud.path || ud.path.length === 0) {
        // Reached destination (or no path) - transition to the next state
        // Ensure position is accurate if a specific destination was set
        if (ud.destino && ud.destino.pos) {
             w.position.set(ud.destino.pos[0], ud.destino.pos[1], ud.destino.pos[2]); // Snap to destination pos
        }
        trocar_state(w, fimState);
        return;
    }

    let targetPoint = new THREE.Vector3(...ud.path[0]);
    let moveDirection = targetPoint.clone().sub(w.position);
    let distanceToTarget = moveDirection.length();

    // Calculate movement for this frame
    let moveDistance = WORKER_SPEED * dt;

    if (moveDistance >= distanceToTarget) {
        // Reached or passed the current waypoint
        w.position.copy(targetPoint); // Snap to waypoint
        ud.path.shift(); // Remove the reached waypoint

        // If that was the last waypoint, transition state
        if (ud.path.length === 0) {
            // Make sure final position matches destination if available
             if (ud.destino && ud.destino.pos) {
                 w.position.set(ud.destino.pos[0], ud.destino.pos[1], ud.destino.pos[2]);
             }
            trocar_state(w, fimState);
        } else {
             // Aim for the next waypoint immediately (recalculate direction)
             let nextTargetPoint = new THREE.Vector3(...ud.path[0]);
             moveDirection = nextTargetPoint.clone().sub(w.position);
             // Rotate to face the next waypoint
             let angle = Math.atan2(moveDirection.x, moveDirection.z); // Note: atan2(x, z) for Y rotation
             w.rotation.y = angle;
        }
    } else {
        // Move towards the current waypoint
        moveDirection.normalize();
        w.position.add(moveDirection.multiplyScalar(moveDistance));

        // Rotate to face the direction of movement
        let angle = Math.atan2(moveDirection.x, moveDirection.z); // Note: atan2(x, z) for Y rotation
         // Smooth rotation?
         // w.rotation.y = THREE.MathUtils.lerp(w.rotation.y, angle, 0.1); // Basic lerp
         w.rotation.y = angle; // Instant rotation
    }
}
